<script>
    let input = "";
    const pass = "500101";

    function press(num) {
        if (input.length < 6) {
            input += num;
            updateDots();
            if (input.length === 6) {
                if (input === pass) {
                    document.getElementById('lock-screen').style.transform = 'translateY(-100%)';
                    setTimeout(() => { document.getElementById('chat-list-page').style.display = 'flex'; }, 500);
                } else {
                    alert("密碼錯誤");
                    reset();
                }
            }
        }
    }

    function updateDots() {
        const dots = document.querySelectorAll('.dot');
        dots.forEach((d, i) => d.classList.toggle('active', i < input.length));
    }

    function reset() {
        input = "";
        updateDots();
    }

    function showChat(chatId) {
        document.getElementById('chat-list-page').style.display = 'none';
        document.getElementById(chatId).style.display = 'flex';
    }

    function backToList() {
        const chats = document.querySelectorAll('.chat-detail');
        chats.forEach(c => c.style.display = 'none');
        document.getElementById('chat-list-page').style.display = 'flex';
    }
</script>
